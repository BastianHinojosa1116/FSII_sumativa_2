import { Injectable } from '@angular/core';

export interface User {
  nombre: string;
  apellido: string;
  nombreUsuario: string;
  correo: string;
  fechaNacimiento: string;
  direccion?: string;
  contrasena: string;
  perfil: string;
}

@Injectable({ providedIn: 'root' })
export class AuthService {
  private storageKey = 'users';
  private sessionKey = 'currentUser';

  private getUsers(): User[] {
  const raw = localStorage.getItem(this.storageKey);
  let users: User[] = raw ? JSON.parse(raw) : [];

  // Si no existe el admin, lo agregamos
  const existeAdmin = users.some(u => u.correo === 'ba.hinojosa@duocuc.cl');
  if (!existeAdmin) {
    const admin: User = {
      nombre: 'Bastian',
      apellido: 'Hinojosa',
      nombreUsuario: 'ba.hinojosa',
      correo: 'ba.hinojosa@duocuc.cl',
      fechaNacimiento: '1990-01-01',
      direccion: 'Oficina Central',
      contrasena: '1Duoc*!',
      perfil: 'administrador'
    };
    users.push(admin);
    localStorage.setItem(this.storageKey, JSON.stringify(users));
  }

  return users;
}


  // 2) Chequea si el correo ya está en uso
  isEmailTaken(email: string): boolean {
    return this.getUsers().some(u => u.correo === email);
  }

  /**
   * 3) Registra un usuario:
   *    - Si el email ya existe, devuelve false.
   *    - Si no, lo añade y devuelve true.
   */
  register(user: User): boolean {
    const users = this.getUsers();
    if (this.isEmailTaken(user.correo)) {
      return false;
    }
    users.push(user);
    localStorage.setItem(this.storageKey, JSON.stringify(users));
    return true;
  }

  login(email: string, password: string): boolean {
    const found = this.getUsers()
      .find(u => u.correo === email && u.contrasena === password);
    if (found) {
      localStorage.setItem(this.sessionKey, JSON.stringify(found));
      return true;
    }
    return false;
  }

  logout(): void {
    localStorage.removeItem(this.sessionKey);
  }

  getCurrentUser(): User | null {
    return JSON.parse(localStorage.getItem(this.sessionKey) || 'null');
  }
}
