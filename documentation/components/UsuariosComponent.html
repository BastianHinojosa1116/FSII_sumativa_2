import { Component, OnInit } from '@angular/core';
import { UsuariosService, Usuario } from '../servicios/usuarios.service';
@Component({
  selector: 'app-usuarios',
  templateUrl: './usuarios.component.html',
  styleUrls: ['./usuarios.component.css']
})
export class UsuariosComponent implements OnInit {
  usuarios: Usuario[] = [];

  constructor(private servicio: UsuariosService) {}

  ngOnInit(): void {
    this.cargarUsuarios();
  }

  cargarUsuarios() {
    this.usuarios = this.servicio.getUsuarios();
  }

  agregarUsuario() {
    const nombre = prompt('Nombre:');
    const apellido = prompt('Apellido:');
    const email = prompt('Email:');
    const password = prompt('Contraseña:');
    const foto = prompt('URL de la foto (opcional):');

    if (!nombre || !apellido || !email || !password) {
      alert('Todos los campos son obligatorios.');
      return;
    }

    const normalizado = email.trim().toLowerCase();
    if (this.usuarios.some(u => u.email.trim().toLowerCase() === normalizado)) {
      alert('Ya existe un usuario con ese email.');
      return;
    }

    const perfil = normalizado === 'bastian1116@hotmail.com' ? 'Administrador' : 'Usuario';

    this.usuarios.push({ nombre, apellido, email, password, foto: foto || '', perfil });
    this.servicio.setUsuarios(this.usuarios);
    this.cargarUsuarios();
  }

  editar(index: number) {
    const u = this.usuarios[index];

    const nombre = prompt('Nuevo nombre:', u.nombre);
    const apellido = prompt('Nuevo apellido:', u.apellido);
    const email = prompt('Nuevo email:', u.email);
    const password = prompt('Nueva contraseña:', u.password);
    const foto = prompt('Nueva URL de foto:', u.foto);

    if (!nombre || !apellido || !email || !password) {
      alert('Todos los campos son obligatorios.');
      return;
    }

    const normalizado = email.trim().toLowerCase();

    if (u.email === 'bastian1116@hotmail.com' && normalizado !== u.email) {
      alert('No puedes cambiar el email del Administrador principal.');
      return;
    }

    if (this.usuarios.some((x, i) => x.email.trim().toLowerCase() === normalizado && i !== index)) {
      alert('Ya existe otro usuario con ese email.');
      return;
    }

    const perfil = normalizado === 'bastian1116@hotmail.com' ? 'Administrador' : u.perfil;

    this.usuarios[index] = { nombre, apellido, email, password, foto: foto || '', perfil };
    this.servicio.setUsuarios(this.usuarios);
    this.cargarUsuarios();
  }

  eliminar(index: number) {
    const u = this.usuarios[index];
    if (u.email === 'bastian1116@hotmail.com') {
      alert('No puedes eliminar al Administrador principal.');
      return;
    }

    if (confirm(`¿Eliminar a ${u.nombre}?`)) {
      this.usuarios.splice(index, 1);
      this.servicio.setUsuarios(this.usuarios);
      this.cargarUsuarios();
    }
  }

  cambiarPerfil(usuario: Usuario, nuevoPerfil: string) {
    if (usuario.email === 'bastian1116@hotmail.com') return;

    usuario.perfil = nuevoPerfil;
    this.servicio.setUsuarios(this.usuarios);
    this.cargarUsuarios();
  }
}
